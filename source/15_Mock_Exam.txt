from abc import ABC, abstractmethod

class Ticket(ABC):
    _nextId = 1

    def __init__(self, movie, basePrice):
        self.__ticketId = Ticket._nextId
        Ticket._nextId += 1
        self.__movie = movie
        self.__basePrice = basePrice

    @property
    def ticketId(self):
        return self.__ticketId

    @property
    def movie(self):
        return self.__movie

    @property
    def basePrice(self):
        return self.__basePrice

    @abstractmethod
    def price(self):
        pass

    def __str__(self):
        return f"#{self.__ticketId} {self.__movie}: ${self.price():.2f}"

class StandardTicket(Ticket):
    _STUDENT_DISCOUNT = 0.2

    def __init__(self, movie, basePrice, isStudent):
        super().__init__(movie, basePrice)
        self.__isStudent = isStudent

    @property
    def isStudent(self):
        return self.__isStudent

    def price(self):
        if self.__isStudent:
            return self.basePrice * (1 - StandardTicket._STUDENT_DISCOUNT)
        return self.basePrice

    def __str__(self):
        if self.__isStudent:
            return super().__str__() + " (Student)"
        return super().__str__()

@@ write Q2(d) Types of overriding (5 marks)
(i) What type of overriding is `PremiumTicket.__str__()`? Explain. (2 marks)

(ii) What type of overriding is `PremiumTicket.price()`? Explain. (2 marks)

(iii) In the sample code, `print(t)` gives a different result for different tickets. What is this OOP idea called? (1 mark)
@@ answer
(i) **Overriding by refinement.**
`PremiumTicket.__str__()` has the same name and parameters as `Ticket.__str__()`, and it calls the parent version with `super().__str__()`, then adds more text.

(ii) **Overriding by replacement.**
`PremiumTicket.price()` overrides `Ticket.price()` but does **not** call `super().price()`.
It provides its own complete calculation (here it implements the abstract method).

(iii) **Polymorphism.**
The same call `print(t)` (which calls `t.__str__()` and `t.price()`) runs a different method depending on the class of the object `t`.

@@ solution
class Booking:
    _MAX_TICKETS = 4
    _MAX_PREMIUM = 2

    def __init__(self, customer):
        self.__customer = customer
        self.__tickets = []

    @property
    def customer(self):
        return self.__customer

    def findTicket(self, ticketId):
        for t in self.__tickets:
            if t.ticketId == ticketId:
                return t
        return None

    def addTicket(self, ticket):
        if self.findTicket(ticket.ticketId) is not None:
            raise CinemaException("Ticket already in booking")
        if len(self.__tickets) + 1 > Booking._MAX_TICKETS:
            raise CinemaException(f"Cannot book more than {Booking._MAX_TICKETS} tickets")
        if isinstance(ticket, PremiumTicket):
            premiumCount = 0
            for t in self.__tickets:
                if isinstance(t, PremiumTicket):
                    premiumCount += 1
            if premiumCount + 1 > Booking._MAX_PREMIUM:
                raise CinemaException(f"Cannot book more than {Booking._MAX_PREMIUM} premium tickets")
        self.__tickets.append(ticket)

class Booking:
    _MAX_TICKETS = 4
    _MAX_PREMIUM = 2

    def __init__(self, customer):
        self.__customer = customer
        self.__tickets = []

    @property
    def customer(self):
        return self.__customer

    def findTicket(self, ticketId):
        for t in self.__tickets:
            if t.ticketId == ticketId:
                return t
        return None

    def addTicket(self, ticket):
        if self.findTicket(ticket.ticketId) is not None:
            raise CinemaException("Ticket already in booking")
        if len(self.__tickets) + 1 > Booking._MAX_TICKETS:
            raise CinemaException(f"Cannot book more than {Booking._MAX_TICKETS} tickets")
        if isinstance(ticket, PremiumTicket):
            premiumCount = 0
            for t in self.__tickets:
                if isinstance(t, PremiumTicket):
                    premiumCount += 1
            if premiumCount + 1 > Booking._MAX_PREMIUM:
                raise CinemaException(f"Cannot book more than {Booking._MAX_PREMIUM} premium tickets")
        self.__tickets.append(ticket)

    def removeTicket(self, ticketId):
        ticket = self.findTicket(ticketId)
        if ticket is None:
            raise CinemaException("Ticket not found")
        self.__tickets.remove(ticket)
        return ticket

    def total(self):
        total = 0
        for t in self.__tickets:
            total += t.price()
        return total

    def __str__(self):
        result = f"Booking for {self.__customer}\n"
        for t in self.__tickets:
            result += f"{t}\n"
        result += f"Total: ${self.total():.2f}"
        return result
